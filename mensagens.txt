t_or=t

function t()
	return t_or()-t_stampd
end

function make_conf()
	return 
	{
		xe=rnd(480),
		ye=rnd(230),
		x=240,
		y=340,
		vy=0,
		c=flr(rnd(14))+1,
		o=rnd(5)+2,
		cair=function(z)
			z.vy+=(1-z.vy)*0.1
			z.y+=z.vy
			z.xv=z.x+z.o*sin(2*3*t()/z.o)
			pset(z.xv,z.y,z.c)
		end,
		subir=function(z)
			z.x+=(z.xe-z.x)*0.1
			z.y+=(z.ye-z.y)*0.1
			z.xv=z.x+z.o*sin(2*3*t()/z.o)
			pset(z.xv,z.y,z.c)
		end
	}
end







function esperar(x)
	for i=1,x do flip() end
end
function escrever(txt,esp,c)
	for i=1,#txt do
		cls()
		sfx(2)
		print(sub(txt,1,i),240-(i*2.5),135,c)
		esperar(4)
	end
	esperar(esp)
end
